from __future__ import annotations

from dataclasses import dataclass
from typing import Any

import numpy as np

from scslp.config import FHOConfig
from scslp.problem import TwoStageProblem


def _safe_int(val: Any) -> int:
    try:
        return int(val)
    except (ValueError, TypeError):
        return 0


def _safe_float(val: Any) -> float:
    try:
        return float(val)
    except (ValueError, TypeError):
        return 0.0


@dataclass
class FHOResult:
    best_solution: np.ndarray
    best_fitness: float
    fitness_history: list[float]
    decoded: dict[str, Any]


class FireHawkOptimizer:
    """Discrete Fire Hawk Optimizer (FHO) metaheuristic for the SCSLP problem."""

    def __init__(self, problem: TwoStageProblem, config: FHOConfig) -> None:
        self.problem = problem
        self.config = config
        self.rng = np.random.default_rng(config.random_seed)

    def _clean_solution(
        self, y_candidate: np.ndarray, n_bikes: int | np.integer[Any], trip_acceptance_prob: float = 0.95
    ) -> np.ndarray:
        """Construct a valid, feasible discrete solution from station decisions and bike count."""
        sol = np.zeros(self.problem.dimension, dtype=int)
        y = np.asarray(y_candidate, dtype=int).copy()

        # Ensure at least one station is built
        if np.sum(y) == 0:
            y[self.rng.integers(0, self.problem.num_stations)] = 1

        # Budget repair for stations
        st_cost = _safe_float(np.dot(self.problem.stations_df["opening_cost"], y))
        while st_cost > self.problem.budget_limit * 0.95 and np.sum(y) > 1:
            open_indices = np.where(y == 1)[0]
            drop_idx = _safe_int(self.rng.choice(open_indices))
            y[drop_idx] = 0
            st_cost = _safe_float(np.dot(self.problem.stations_df["opening_cost"], y))

        sol[: self.problem.num_stations] = y

        # System capacity & bike count
        open_cap = _safe_int(np.dot(self.problem.capacities, y))
        max_bikes_budget = _safe_int((self.problem.budget_limit - st_cost) // self.problem.bike_cost)
        actual_bikes = max(5, min(n_bikes, open_cap, max_bikes_budget))
        if actual_bikes > 0:
            sol[self.problem.dim_y : self.problem.dim_y + actual_bikes] = 1

        # Trip acceptance based on station connectivity
        x = np.zeros((self.problem.num_scenarios, self.problem.max_trips), dtype=int)
        for sc in range(self.problem.num_scenarios):
            origs = self.problem.scenario_trip_origins[sc]
            dests = self.problem.scenario_trip_destinations[sc]
            num_t = min(len(origs), self.problem.max_trips)
            for t in range(num_t):
                if (
                    y[origs[t]] == 1
                    and y[dests[t]] == 1
                    and self.rng.random() < trip_acceptance_prob
                ):
                    x[sc, t] = 1

        sol[self.problem.dim_y + self.problem.dim_b :] = x.flatten()
        return sol

    def _initialize_population(self) -> np.ndarray:
        """Create initial diverse population with varying station counts and bike deployments."""
        pop = []
        for _ in range(self.config.pop_size):
            k_st = self.rng.integers(6, self.problem.num_stations + 1)
            y_init = np.zeros(self.problem.num_stations, dtype=int)
            selected_st = self.rng.choice(self.problem.num_stations, size=k_st, replace=False)
            y_init[selected_st] = 1
            bikes_init = _safe_int(self.rng.integers(15, 60))
            pop.append(self._clean_solution(y_init, bikes_init, trip_acceptance_prob=0.9))
        return np.array(pop)

    def optimize(self) -> FHOResult:
        """Execute Discrete Fire Hawk Optimizer iterations."""
        pop = self._initialize_population()
        fitnesses = np.array([self.problem.evaluate(ind) for ind in pop])

        best_idx = _safe_int(np.argmax(fitnesses))
        best_sol = pop[best_idx].copy()
        best_fit = _safe_float(fitnesses[best_idx])
        history: list[float] = [best_fit]

        num_fh = max(1, min(self.config.num_fire_hawks, self.config.pop_size // 2))

        for _ in range(self.config.max_generations - 1):
            # Sort population descending by fitness
            order = np.argsort(fitnesses)[::-1]
            pop = pop[order]
            fitnesses = fitnesses[order]

            fire_hawks = pop[:num_fh].copy()
            preys = pop[num_fh:].copy()

            # 1. Fire Hawks guided search
            for i in range(num_fh):
                y_fh = fire_hawks[i][: self.problem.num_stations].copy()
                b_fh = _safe_int(np.sum(fire_hawks[i][self.problem.dim_y : self.problem.dim_y + self.problem.dim_b]))

                # Guided crossover towards global best
                crossover_mask = self.rng.random(self.problem.num_stations) < 0.45
                y_fh[crossover_mask] = best_sol[: self.problem.num_stations][crossover_mask]

                # Bounded mutation on station selection
                if self.rng.random() < 0.35:
                    mut_idx = self.rng.integers(0, self.problem.num_stations)
                    y_fh[mut_idx] = 1 - y_fh[mut_idx]

                # Perturb bike allocation
                b_fh = _safe_int(max(5, b_fh + self.rng.integers(-3, 4)))
                candidate = self._clean_solution(y_fh, b_fh, trip_acceptance_prob=0.98)
                fit_c = self.problem.evaluate(candidate)

                if fit_c > fitnesses[i]:
                    fire_hawks[i] = candidate
                    fitnesses[i] = fit_c

            # 2. Preys territory update
            new_preys = []
            for j in range(len(preys)):
                y_prey = preys[j][: self.problem.num_stations].copy()
                b_prey = _safe_int(np.sum(preys[j][self.problem.dim_y : self.problem.dim_y + self.problem.dim_b]))

                # Follow assigned territory Fire Hawk
                target_fh = fire_hawks[self.rng.integers(0, num_fh)]
                follow_mask = self.rng.random(self.problem.num_stations) < 0.50
                y_prey[follow_mask] = target_fh[: self.problem.num_stations][follow_mask]

                # Safe Place (elite guide) pull
                if self.rng.random() < 0.20:
                    sp_mask = self.rng.random(self.problem.num_stations) < 0.30
                    y_prey[sp_mask] = best_sol[: self.problem.num_stations][sp_mask]

                # Exploration mutation
                if self.rng.random() < 0.25:
                    mut_idx = self.rng.integers(0, self.problem.num_stations)
                    y_prey[mut_idx] = 1 - y_prey[mut_idx]

                b_prey = _safe_int(max(5, b_prey + self.rng.integers(-4, 5)))
                candidate_prey = self._clean_solution(y_prey, b_prey, trip_acceptance_prob=0.95)
                fit_p = self.problem.evaluate(candidate_prey)

                if fit_p > fitnesses[num_fh + j]:
                    new_preys.append(candidate_prey)
                    fitnesses[num_fh + j] = fit_p
                else:
                    new_preys.append(preys[j])

            pop = np.vstack([fire_hawks, new_preys])
            curr_best = _safe_float(np.max(fitnesses))
            if curr_best > best_fit:
                best_fit = curr_best
                best_sol = pop[_safe_int(np.argmax(fitnesses))].copy()

            history.append(best_fit)

        return FHOResult(
            best_solution=best_sol,
            best_fitness=best_fit,
            fitness_history=history,
            decoded=self.problem.decode_solution(best_sol),
        )
